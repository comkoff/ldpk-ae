116  vec2_type eval(const vec2_type& p_dn) const
117  {
118  double x_dn,y_dn;
119  double x = p_dn[0];
120  double y = p_dn[1];
121  double x2 = x * x;
122  double y2 = y * y;
123  double xy = x * y;
124  double r2 = x2 + y2;
125  double r4 = r2 * r2;
126  double r6 = r2 * r4;
127 // OpenCV has Y-down-coordinates
128  x_dn = x * (1.0 + _c2 * r2 + _c4 * r4 + _c6 * r6)
129  + (r2 + 2.0 * x2) * (_u2)
130  + 2.0 * xy * (-_v2);
131 
132  y_dn = y * (1.0 + _c2 * r2 + _c4 * r4 + _c6 * r6)
133  + (r2 + 2.0 * y2) * (-_v2)
134  + 2.0 * xy * (_u2);
135  return vec2_type(x_dn,y_dn);
136  }
139  mat2_type jacobi(const vec2_type& p_dn) const
140  {
141  mat2_type m;
142  double x = p_dn[0];
143  double y = p_dn[1];
144  double x2 = x * x;
145  double y2 = y * y;
146  double x3 = x2 * x;
147  double y3 = y2 * y;
148  double xy = x * y;
149  double x2y = xy * x;
150  double xy2 = xy * y;
151  double r2 = x2 + y2;
152  double r4 = r2 * r2;
153 
154 // long implementation for testing.
155 // m[0][0] = 1 + _c2 * (3.0 * x2 + y2) + _c4 * (5.0 * x2 + y2) * r2 + _c6 * (7.0 * x2 + y2) * r4
156 // + 6.0 * _u2 * x
157 // + 2.0 * (-_v2) * y;
158 // m[1][1] = 1 + _c2 * (x2 + 3.0 * y2) + _c4 * (x2 + 5.0 * y2) * r2 + _c6 * (x2 + 7.0 * y2) * r4
159 // + 6.0 * (-_v2) * y
160 // + 2.0 * _u2 * x;
161 //
162 // m[0][1] = 2.0 * _c2 * xy + 4.0 * _c4 * xy * r2 + _c6 * xy * r4
163 // + 2.0 * _u2 * y
164 // + 2.0 * (-_v2) * x;
165 // m[1][0] = 2.0 * _c2 * xy + 4.0 * _c4 * xy * r2 + _c6 * xy * r4
166 // + 2.0 * _u2 * y
167 // + 2.0 * (-_v2) * x;
168 // slightly condensed implementation
169  double u2x = _u2 * x;
170  double v2y = (-_v2) * y;
171  double c4r2 = _c4 * r2;
172  double c6r4 = _c6 * r4;
173  m[0][0] = 1.0 + _c2 * (3.0 * x2 + y2) + c4r2 * (5.0 * x2 + y2) + c6r4 * (7.0 * x2 + y2)
174  + 6.0 * u2x
175  + 2.0 * v2y;
176  m[1][1] = 1.0 + _c2 * (x2 + 3.0 * y2) + c4r2 * (x2 + 5.0 * y2) + c6r4 * (x2 + 7.0 * y2)
177  + 6.0 * v2y
178  + 2.0 * u2x;
179 
180  double m_off_diag_common = 2.0 * _c2 * xy + 4.0 * _c4 * xy * r2 + 6.0 * _c6 * xy * r4
181  + 2.0 * _u2 * y + 2.0 * (-_v2) * x;
182  m[0][1] = m_off_diag_common;
183  m[1][0] = m_off_diag_common;
184 
185  return m;

190  void derive(double* dg,int n_parameters,const vec2_type& p_dn) const
191  {
192  int size = 2 * n_parameters;
193  double x = p_dn[0];
194  double y = p_dn[1];
195  double x2 = p_dn[0] * p_dn[0];
196  double y2 = p_dn[1] * p_dn[1];
197  double xy = p_dn[0] * p_dn[1];
198  double r2 = x2 + y2;
199  double r4 = r2 * r2;
200  double r6 = r2 * r4;
201 // OpenCV order. Since we have defined c_sdv = c_ocv * rho**j,
202 // the derivatives d/dc_ocv are d/dc_sdv * rho**j.
203  int k = 0;
204 // c2 (= k1)
205  dg[k++] = _rho_pow[0] * (x * r2);
206  dg[k++] = _rho_pow[0] * (y * r2);
207  if(k == size) return;
208 // c4 (= k2)
209  dg[k++] = _rho_pow[1] * (x * r4);
210  dg[k++] = _rho_pow[1] * (y * r4);
211  if(k == size) return;
212 // v2 (= p1), mind the sign due to OpenCV's Y-down-coordinates.
213  dg[k++] = -_rho_pow[2] * (2.0 * xy);
214  dg[k++] = -_rho_pow[2] * (r2 + 2.0 * y2);
215  if(k == size) return;
216 // u2 (= p2)
217  dg[k++] = _rho_pow[3] * (r2 + 2.0 * x2);
218  dg[k++] = _rho_pow[3] * (2.0 * xy);
219  if(k == size) return;
220 // c6 (= k3)
221  dg[k++] = _rho_pow[4] * (x * r6);
222  dg[k++] = _rho_pow[4] * (y * r6);
223  if(k == size) return;
224 // Unreachable
225  std::cerr << "radial_decentered_distortion_opencv: n_parameters out of range" << std::endl;
226  }
