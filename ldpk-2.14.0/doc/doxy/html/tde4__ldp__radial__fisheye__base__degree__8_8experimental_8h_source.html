232  bool undistort(double x0,double y0,double &x1,double &y1) override
233  {
234  typedef base_type bt;
235 
236  vec2_type p_any_dist_dn = bt::map_unit_to_dn(vec2_type(x0,y0));
237 
238 // Distance from origin in dn-coordinates
239  double r_any_dist_dn = norm2(p_any_dist_dn);
240 // We have determined the domain of undistort() in initializeParameters().
241  if(r_any_dist_dn > _r_any_dn_domain)
242  { return false; }
243 // The literature commonly says, undistorting should be done
244 // before applying the remapping. Usually it's applied to
245 // elongation angle theta. This is what we do here: We map
246 // the radial position to the corresponding polar angle,
247 // undistort the angle and map it to a radius in planar coordinates.
248 // This guarantees that scale only affects the domain size, not the shape
249 // of the distortion.
250  double theta_dist;
251  if(!_mapfun.map_any2angle(r_any_dist_dn,theta_dist))
252  { return false; }
253  double theta_undist = _distortion.eval(theta_dist);
254 // Remap angle to gnomonic
255  double r_plain_undist_dn;
256  if(!_mapfun.map_angle2plain(theta_undist,r_plain_undist_dn))
257  { return false; }
258 
259 // Calc undistorted point from undistorted radius
260  vec2_type p_plain_undist_dn;
261  if(dotsq(p_any_dist_dn) == 0.0)
262  { p_plain_undist_dn = vec2_type(0,0); }
263  else
264  { p_plain_undist_dn = r_plain_undist_dn * unit(p_any_dist_dn); }
265 // Map back to unit coordinates
266  vec2_type p_plain_undist_unit = bt::map_dn_to_unit(p_plain_undist_dn);
267  x1 = p_plain_undist_unit[0];
268  y1 = p_plain_undist_unit[1];
269  return true;

273  typedef base_type bt;
274 
275  vec2_type p_plain_undist_dn = bt::map_unit_to_dn(vec2_type(x0,y0));
276 // Distance from origin in dn-coordinates
277  double r_plain_undist_dn = norm2(p_plain_undist_dn);
278 // We have determined the domain of undistort() in initializeParameters().
279 // Using the undistorted radius, we can formulate a criterion for distort().
280  if(r_plain_undist_dn > _r_plain_dn_domain)

351  bool getJacobianMatrix(double x0, double y0, double &m00, double &m01, double &m10, double &m11) override
352  {
353  typedef base_type bt;
354 
355  vec2_type p_dn = bt::map_unit_to_dn(vec2_type(x0,y0));
356  double norm_p_dn = norm2(p_dn);
357  vec2_type unit_p_dn = unit(p_dn);
358  mat2_type tensq_unit_p_dn = tensq(unit_p_dn);
359 // According to our document we need several intermediate quantities.
360 // The distorted angle.
361  double F_inv_at_norm_p_dn;
362  if(!_mapfun.map_any2angle(norm_p_dn,F_inv_at_norm_p_dn))
363  { return false; }
364 // The undistorted angle.
365  double g_poly_rad_at_theta_dist = _distortion.eval(F_inv_at_norm_p_dn);
366 // The remapped radius
367  double G_at_g_poly_rad_at_theta_dist;
368  _mapfun.map_angle2plain(g_poly_rad_at_theta_dist,G_at_g_poly_rad_at_theta_dist);
369 // The derivative of the inverse mapping function at |p|.
370  double dF_inv_by_dr_dn;
371  if(!_mapfun.derive_any2angle(norm_p_dn,dF_inv_by_dr_dn))
372  { return false; }
373 // The derivative of the distortion polynomial at F_inv_at_norm_p_dn.
374  double dg_poly_rad_by_theta = _distortion.jacobi(F_inv_at_norm_p_dn);
375 // The derivative of the remapping function at g_poly_rad_at_theta_dist.
376  double dG_by_theta = _mapfun.derive_angle2plain(g_poly_rad_at_theta_dist);
377 // Apply the Leibniz chain rule
378  double dg_any_rad_by_dr = dG_by_theta * dg_poly_rad_by_theta * dF_inv_by_dr_dn;
379 // Extend to 2-dim space.
380  mat2_type m;
381  if(norm_p_dn == 0.0)
382  { m = mat2_type(1.0); }
383  else
384  { m = 1.0 / norm_p_dn * (mat2_type(1.0) - tensq_unit_p_dn) * G_at_g_poly_rad_at_theta_dist + tensq_unit_p_dn * dg_any_rad_by_dr; }
385 
386 // unit to diagnorm and vice versa as transforms for the Jacobian.
387  mat2_type u2d(bt::w_fb_cm() / bt::r_fb_cm(),0.0,0.0,bt::h_fb_cm() / bt::r_fb_cm());
388  mat2_type d2u(bt::r_fb_cm() / bt::w_fb_cm(),0.0,0.0,bt::r_fb_cm() / bt::h_fb_cm());
389  m = d2u * m * u2d;
390  m00 = m[0][0];m01 = m[0][1];m10 = m[1][0];m11 = m[1][1];
391  return true;
392  }
