113 // The distort-method without initial values is not constant by semantics,
114 // since it may cause an update of the lookup-tables. Implementing a Nuke node

116 // By the following double check of is_uptodate_lut() we keep the mutex lock
117 // out of our frequently called distort stuff (for performance reasons) and
118 // prevent threads from updating without need.
119  if(!bt::is_uptodate_lut())

121  bt::lock();
122  if(!bt::is_uptodate_lut())

124  bt::update_lut();
125  }
126  bt::unlock();
127  }
128

133  _radial.eval_inv(

135  bt::map_unit_to_dn(qs)));
