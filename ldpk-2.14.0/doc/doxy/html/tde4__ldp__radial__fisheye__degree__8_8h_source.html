395  typedef base_type bt;
396 // The function we need to derive is a concatenation of a simple radially symmetric
397 // undistort funtion and the equisolid-angle transform. We compute this by Leipnitz's rule.
398 // Exterior derivative:
399  vec2_type p_esa_dn = bt::map_unit_to_dn(vec2_type(x0,y0));
400  vec2_type p_plain_dn;
401  if(!remap_esa2plain(p_esa_dn,p_plain_dn))
402  {
403  m00 = 1;m01 = 0;m10 = 0;m11 = 1;
404  return false;
405  }
406  mat2_type m = _distortion.jacobi(p_plain_dn);
407 // Interior derivative
408  mat2_type e;
409  double f_dn = this->fl_cm() / this->r_fb_cm();
410  double r_esa_dn = norm2(p_esa_dn);
411  double r_esa_div_2f = r_esa_dn / (2.0 * f_dn);
412 // Test (a): make sure argument of arcsine is valid.
413  if(fabs(r_esa_div_2f) > 0.99)
414  {
415  m00 = 1;m01 = 0;m10 = 0;m11 = 1;
416  return false;
417  }
418  double asn = asin(r_esa_div_2f);
419 // Test (b): make sure cos(..) is greater than 0.
420  if(fabs(2.0 * asn) > 0.99 * M_PI / 2.0)
421  {
422  m00 = 1;m01 = 0;m10 = 0;m11 = 1;
423  return false;
424  }
425  double csn = cos(2.0 * asn);
426 // Argument of square root is positive because of test (a).
427  double sqr = sqrt(1.0 - r_esa_div_2f * r_esa_div_2f);
428  if(r_esa_dn > 1e-12)
429  {
430 // The analytic expression is undefined at 0...
431  e = (mat2_type(1) - tensq(p_esa_dn / r_esa_dn)) * (f_dn / r_esa_dn) * tan(2.0 * asn)

433  }
434  else
435  {
436 // ...but has well-defined limit
437  e = mat2_type(1);
438  }
