172  {
173  int ix,iy;
174  calc_index(p_fov,ix,iy);
175 // If the index pair is outside the range covered by the lookup table,
176 // we shall not use the table for extracting initial values.
177  if((ix < 0) || (ix >= _nx) ||(iy < 0) || (iy >= _ny))
178  {
179  clipped = true;
180 // We project the point to the fov (in fov coordinates). With 0.5 * unitinf the point
181 // should lie exactly on the fov boundary. The caller may want to rescale the point
182 // further in order to find a closer match. In fact we're doing a 5*5 brute force
183 // scan around the proposed initial value.
184  return 0.5 * unitinf(p_fov - vec2_type(.5,.5)) + vec2_type(.5,.5);
185  }
186 // While a negative Jacobian determinant tells us reliably that the initial value
187 // is unusable, a positive one does not guarantee that they are usable. Nonetheless
188 // this has proved to be an appropriate means to rule out table entries:
189  else if(_samples[ix + _nx * iy].det_j <= 0.0)

191  clipped = true;
