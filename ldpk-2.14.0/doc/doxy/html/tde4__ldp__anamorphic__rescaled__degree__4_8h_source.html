143  bool distort(double x0,double y0,double &x1,double &y1) override
144  {
145 // Implementing a Nuke node it turned out that we need to prevent
146 // threads from trying so simultaneously. By the following double
147 // check of is_uptodate_lut() we keep the mutex lock out of our
148 // frequently called distort stuff (for performance reasons) and
149 // prevent threads from updating without need.
150  if(!this->is_uptodate_lut())
151  {
152  this->lock();
153  if(!this->is_uptodate_lut())
154  { this->update_lut(); }
155  this->unlock();
156  }
157 // Get initial value from lookup-table
158  bool clipped;
159  vec2_type q(x0,y0);
160  vec2_type ps = this->get_lut().get_initial_value(q,clipped);
161 // Call inverse evaluation with initial value.
162  vec2_type q_ana_dn =
163  _rot_sqx_sqy_rescale_pa.eval_inv(
164  this->map_unit_to_dn(q));
165 // Initial value for p in dn-coordinates. Since we feed the
166 // initial value from the lookup table into the model function
167 // we have to unroll the last two function applications.
168  vec2_type ps_dn =
169  _pa_rescale_rot.eval_inv(
170  this->map_unit_to_dn(ps));
171 
172  vec2_type p;
173  if(clipped)
174  { p = this->map_dn_to_unit(
175  _pa_rescale_rot.eval(
176  _anamorphic.eval_inv_clipped(q_ana_dn,ps_dn))); }
177  else
178  { p = this->map_dn_to_unit(
179  _pa_rescale_rot.eval(
180  _anamorphic.eval_inv(q_ana_dn,ps_dn))); }
181  x1 = p[0];
182  y1 = p[1];
183  return true;
184  }
