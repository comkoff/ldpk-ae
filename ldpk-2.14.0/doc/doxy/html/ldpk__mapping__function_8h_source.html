37 // The mapping functions are base on a focal length in diagonally
38 // normalized coordinates. This does not necessarily coincide with

46 // the scale is applied to the fisheye side, i.e. there
47 // is a direct impact on the disc that represents the
48 // target space of remap_plain2any. This parameter corresponds
49 // to the scale parameter we have in the distortion models.

276  mat2_type jacobi(const vec2_type& p_any) const
277  {
278  double r_plain,dr_plain;
279  double r_any = norm2(p_any);
280 // Mapping to Cartesian coordinates introduces
281 // a definition gap, which we solve here.
282  if(r_any == 0.0)
283  { return mat2_type(1.0); }
284  _valid = remap_any2plain(r_any,r_plain);
285  if(!_valid)
286  {
287  _clipped = false;
288  return mat2_type(0.0);
289  }
290  if(!_clipped)
291  {
292  _valid = derive_any2plain(r_any,dr_plain);
293  }
294  else
295  {
296 // If the result has been clipped, we compute the Jacobian
297 // for the clipped position. Not tested.
298  double r_any_clipped;
299  remap_plain2any(_r_plain_dn_clip,r_any_clipped);
300  _valid = derive_any2plain(r_any_clipped,dr_plain);
301  }
302  if(!_valid)
303  {
304  _clipped = false;
305  return mat2_type(0.0);
306  }
307  mat2_type tensq_p_any = tensq(unit(p_any));
308  return (mat2_type(1.0) - tensq_p_any) / r_any * r_plain + tensq_p_any * dr_plain;
309  }
